{%- comment -%} Normalise: only a paragraph that is entirely bold counts as an item headline. A paragraph that merely starts with bold text gets a leading space so it is not mistaken for one. Works on the variable `body` and re-assigns it. {%- endcomment -%} {%- assign _ps = body | split: "

" -%} {%- assign _out = _ps[0] -%} {%- for _p in _ps offset: 1 -%} {%- assign _t = _p | split: "

" | first | strip -%} {%- assign _starts = _t | slice: 0, 8 -%} {%- assign _ends = _t | slice: -9, 9 -%} {%- if _starts == "" and _ends != "" -%} {%- assign _q = _p | replace_first: "", " " -%} {%- else -%} {%- assign _q = _p -%} {%- endif -%} {%- assign _out = _out | append: "

" | append: _q -%} {%- endfor -%} {%- assign body = _out -%}
